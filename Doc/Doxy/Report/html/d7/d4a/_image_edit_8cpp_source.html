196/*!
197 *
198 * This function is called after the 'Apply' button is pressed, copyCrop
199 * is active and a valid cropping rectangle is selected. Then it copies
200 * the selected region into a new image with the original name and an
201 * increasing number like (1), (2).
202 *
203 */
204
205void ImageEdit::copyImgRect ()
206{
207 Mat cropped;
208 Mat * lastImg;
209 QString imgName;
210 QString newName;
211 QString help;
212 QString ext;
213 QString path;
214 int pos;
215
216 lastImg = toolsWidget->getLastImage ();
217
218 cropped = imageTools.cropImage ( * lastImg, false );
219
220 imgName = imageWidget->getImageName ();
221
222 pos = imgName.lastIndexOf ( '.' );
223 path = imgName.left (pos);
224 ext = imgName.mid (pos);
225
226 // create new unique file name
227 for ( int i = 0; i < 50; i++ )
228 {
229 help = QString ( "_(%1)" ).arg ( i + 1, 2, 10, QChar ( '0' ) );
230 newName = path + help + ext;
231
232 if ( not QFile::exists (newName) )
233 break;
234 }
235
236 toolsWidget->saveImage ( newName, cropped );
237
238 text = "Cropped image saved as " + newName;
239 mainWnd->updateStatusBar ( "Info", text );
240

244/*!
245 *
246 * This function calls 'getContourPts' to detect the outer vertices of
247 * all contours found in the image. It returns the points of the rotated
248 * rectangle of the minimum area enclosing the points of all contours.
249 * The vertices are then sorted so that have the order : top left, top
250 * right, bottom left, bottom right
251 *
252 * \param inMat
253 * \param vertices
254 *
255 */
256
257void ImageEdit::getVertices ( Mat inMat, L_PointsF & vertices )
258{
259 V_CvPoint points;
260 V_CvPoint vertPtns;
261 QPointF point;
262
263 imageTools.getContourPts ( inMat, points );
264
265 // sort and reduce to 4 points
266 vertPtns = imageTools.getDocPoints ( image->rows, image->cols, points );
267
268 vertices.clear ();
269
270 for ( qint i = 0; i < vertPtns.size (); i++ )
271 {
272 point.setX ( vertPtns [i].x );
273 point.setY ( vertPtns [i].y );
274
275 vertices.append (point);
276 }
277
278}

281/*!
282 *
283 * This function returns the export directory, it is taken from the path
284 * select combobox.
285 *
286 * \return QString
287 *
288 */
289
290QString ImageEdit::getCropRect ()
291{
292 L_String margins;
293 QRectF rect;
294 int cols;
295 int rows;
296 float margin;
297
298 rect = imageScene->getRect ();
299
300 margins.clear ();
301
302 rows = image->rows;
303 cols = image->cols;
304
305 margin = 1.0 - ( rows - rect.top () ) / rows;
306 text.setNum ( 100.0 * margin, 'g', 3 );
307 margins.append (text);
308
309 margin = 1.0 * ( rows - rect.bottom () ) / rows;
310 text.setNum ( 100.0 * margin, 'g', 3 );
311 margins.append (text);
312
313 margin = 1.0 - ( cols - rect.left () ) / cols;
314 text.setNum ( 100.0 * margin, 'g', 3 );
315 margins.append (text);
316
317 margin = 1.0 * ( cols - rect.right () ) / cols;
318 text.setNum ( 100.0 * margin, 'g', 3 );
319 margins.append (text);
320
321 text = margins.join ( "," );
322
323 return text;
324
325}

328/*!
329 *
330 * This function is called when the 'Ok' button is pressed. It does not
331 * reset the points in the level curve before it closes the dialog.
332 *
333 */
334
335void ImageEdit::onApply ()
336{
337 AppConfig configBak;
338 QLineF line;
339 double angle;
340
341 mainWnd->fromGui (); // calls fromGui also from toolsWidget and self
342
343 configBak = * appConfig; // make a copy
344
345 switch (activeMode)
346 {
347 case cInsertNone:
348 return;
349 break;
350
351 case cInsertLine:
352 {
353 line = imageScene->getLine ();
354 angle = line.angle (); // angle is measured counter-clockwise
355
356 if ( angle > 180.0 )
357 angle = angle - 360.0;
358
359 toolsWidget->setRotation ( - angle ); // rotate in opposite direction
360
361 appConfig->rotation = -angle;
362 appConfig->useRotate = true; // use angle from gui
363 appConfig->useAutoRot = false; // no autorotation
364
365 if ( appConfig->runToolChain )
366 toolsWidget->runChainTo (cRunRotate); // apply including rotate
367 else
368 toolsWidget->runTool (cRunRotate); // run only rotate
369 }
370 break;
371
372 case cInsertRect:
373 {
374 text = getCropRect ();
375
376 appConfig->cropMargins = text;
377 appConfig->useCropImg = true; // use page margins from gui
378 appConfig->useAutoCrop = false; // no autorotation
379
380 configBak.cropMargins = appConfig->cropMargins; // store also in backup
381
382 if (cropCopy)
383 {
384 if ( appConfig->runToolChain )
385 toolsWidget->runChainTo ( cRunCrop - 1 ); // apply without crop
386
387 copyImgRect ();
388 }
389 else
390 {
391 if ( appConfig->runToolChain )
392 toolsWidget->runChainTo (cRunCrop); // apply including crop
393 else
394 toolsWidget->runTool (cRunCrop); // run only crop
395 }
396 }
397 break;
398
399 case cInsertPath:
400 {
401 appConfig->useDeskew = true;
402
403 if ( appConfig->runToolChain )
404 toolsWidget->runChainTo (cRunDeskew); // apply including deskew
405 else
406 toolsWidget->runTool (cRunDeskew); // run only deskew
407 }
408 break;
409 }
410
411 * appConfig = configBak; // copy back original settings
412
413 onCancel (); // finish current action
414
415}

479/*!
480 *
481 * This function is called when the 'Deskew' button is pressed. It starts
482 * a path selection with 4 points. After applying the action it applies a
483 * perspective transformation of image to the selected path.
484 *
485 */
486
487void ImageEdit::onDeskew ()
488{
489 L_PointsF points;
490 QRectF rect;
491
492 if ( not prepareEdit ( cRunDeskew - 1 ) )
493 return;
494
495 activeMode = cInsertPath;
496
497 points.clear ();
498
499 if ( appConfig->autoDetect )
500 {
501 getVertices ( * image, points );
502 }
503
504 if ( ( not appConfig->autoDetect ) or ( points.size () != 4 ) )
505 {
506 rect.setX ( 0.1 * image->cols );
507 rect.setY ( 0.1 * image->rows );
508 rect.setWidth ( 0.8 * image->cols );
509 rect.setHeight ( 0.8 * image->rows );
510
511 points.clear ();
512 points.append ( rect.topLeft () );
513 points.append ( rect.topRight () );
514 points.append ( rect.bottomLeft () );
515 points.append ( rect.bottomRight () );
516 }
517
518 insertMode = cMovePath; // and start with moving the lines
519
520 imageScene->makePath (points); // make the line in image scene
521
522 imageScene->setInsertMode (insertMode);
523
524}

572void ImageEdit::onCrop ()
573{
574 QRectF rect;
575 cv::Rect box;
576
577 if ( not prepareEdit ( cRunCrop - 1 ) )
578 return;
579
580 if (cropCopy)
581 imageScene->setPenColor ( Color::darkgreen );
582
583 activeMode = cInsertRect;
584
585 if ( appConfig->autoDetect )
586 {
587 box = imageTools.getContourRect ( * image );
588

591 rect.setX ( box.x - 3 ); // add some pixel margin
592 rect.setY ( box.y - 3 );

594 rect.setHeight ( box.height + 6 );

596 else
597 {
598 rect.setX (5); // add some pixel margin
599 rect.setY (5);
600 rect.setWidth ( image->cols - 10 );
601 rect.setHeight ( image->rows - 10 );
602 }
603
604 insertMode = cMoveRect; // and start with moving the lines
605
606 imageScene->makeRect (rect); // make the line in image scene
607 }
608 else

610 insertMode = cInsertRect;

649/*!
650 *
651 * This function is called when the 'Rotate' button is pressed. It allows
652 * to visually draw a line with with two end points. The angle of the
653 * line can then be applied to the image. If autodetect is checked then
654 * an automatic rotation analysis of the image is done first and a
655 * default line with the found rotation is created.
656 *
657 */
658
659void ImageEdit::onRotate ()
660{
661 QLineF line;
662 QPointF p1;
663 QPointF p2;
664 float angle;
665
666 if ( insertMode != cInsertNone )
667 onCancel (); // cancel last action
668
669 mainWnd->fromGui ();
670
671 if ( appConfig->runToolChain )
672 toolsWidget->runChainTo (cRunDeskew); // apply including deskew
673
674 image = toolsWidget->getLastImage ();
675
676 activeMode = cInsertLine;
677
678 imageScene->removeItems ();
679
680 imageScene->setPenColor ( Color::blue );
681 imageScene->setPenWidth ( 5.0 );
682
683 if ( appConfig->autoDetect )
684 {
685 // OpenCV counts angle clockwise, change to counter-clockwise
686 angle = - imageTools.detectAngle ( * image );
687
688 text.setNum ( angle, 'g', 3 );
689 mainWnd->updateStatusBar ( "Info", "Angle: " + text + " deg" );
690
691 toolsWidget->setRotation ( -angle ); // rotate in opposite direction
692
693 // create a default line with the given angle
694
695 p1.setX ( image->cols * 0.2 ); // point 1 at 20 % of the image width
696 p1.setY ( image->rows * 0.2 ); // point 1 at 20 % of the image height
697
698 p2.setX ( image->cols * 0.8 ); // point 2 at 80 % of the image width
699 p2.setY ( image->rows * 0.2 ); // point 2 at 20 % of the image height
700
701 line.setP1 (p1);
702 line.setP2 (p2);
703
704 line.setAngle (angle); // counts angle counter-clockwise
705
706 p2 = line.p2 ();
707
708 imageScene->makeLine ( p1, p2 ); // make the line in image scene
709 insertMode = cMoveLine; // and start with moving the line
710 }
711 else
712 {
713 insertMode = cInsertLine;
714 }
715
716 imageScene->setInsertMode (insertMode);
717
718}
