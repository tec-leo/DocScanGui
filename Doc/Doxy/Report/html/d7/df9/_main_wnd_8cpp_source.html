234 *
235 * This function creates the parameter manager and reads the last
236 * configuration file from the global config or creates a default if
237 * there is no valid configuration. Then it reads the settings from the
238 * config file and stores the pointer for the config files.

241
242void MainWnd::createSettings ()
243{
244 FileUtils fileUtils;
245 QString fileName;
246
247 fileName = RegFile::getFilePath () + "/GlobalConfig.ini";
248 regFile = new RegFile (fileName);
249
250 defaultIni = fileUtils.absolutePath (fileName) + "/DefaultConfig.ini";
251
252 // create the parameter manager, must be one of the first steps
253 parMan = new ParManager ();
254
255 // in command mode configuration file name could already be set
256 if ( configName.isEmpty () )
257 {
258 // check if another file name is in the ini file
259 if ( regFile->contains ( "LastConfig" ) )
260 {
261 text = regFile->value ( "LastConfig" ).toString ();
262
263 if ( QFile::exists (text) )
264 configName = text;
265 }
266 else
267 configName = defaultIni;
268 }
269
270 // if file does not exist or control key is pressed open a select dialog
271 if ( not QFile::exists (configName) )
272 {
273 if ( not fileUtils.createFile (configName) ) // create if possible
274 {
275 text = "Cannot create config file " + configName +
276 ",\nusing " + defaultIni + " as default";
277 msgLog.showMessage ( "Config file", text, " from Main Window" );
278
279 configName = defaultIni;
280 fileUtils.createFile (configName);
281 }
282 }
283
284 parMan->init (configName);
285
286 appConfig = parMan->getAppConfig ();
287 fileConfig = parMan->getFileConfig ();

501/*!
502 *
503 * This function is called from 'executeJob' evaluates the command for
504 * configuration jobs like load, save, save curve and set options.
505 *
506 */
507
508void MainWnd::execConfig ()
509{
510 L_String keys;
511 QString name;
512 QString key;
513 QString entry;
514
515 if ( jobInfo.command == "load-curve" )
516 {
517 if ( infoMap.contains ( "curve-name" ) )
518 {
519 name = infoMap [ "curve-name" ];
520
521 parMan->getAppCurves ()->selectCurve (name);
522 }
523 }
524
525 if ( jobInfo.command == "load-config" )
526 {
527 if ( infoMap.contains ( "config-file" ) )
528 {
529 name = infoMap [ "config-file" ];
530
531 if ( QFile::exists (name) )
532 parMan->load (name);
533 }
534 }
535
536 if ( jobInfo.command == "save-config" )
537 {
538 if ( infoMap.contains ( "config-file" ) )
539 {
540 name = infoMap [ "config-file" ];
541 parMan->setFileName (name);
542 }
543 else
544 name = parMan->getFileName ();
545
546 if ( not fileUtils.makeFilePath (name) )
547 {
548 text = "Cannot create directory for " + name;
549 msgLog.showError ( "Save config", text, " from Main Window" );
550
551 return;
552 }
553
554 parMan->save ();
555 }
556
557 if ( jobInfo.command == "set-options" )
558 {
559 keys = infoMap.keys ();
560
561 for ( int i = 0; i < keys.count (); i++ )
562 {
563 key = keys [i];
564
565 if ( ( key == "topic" ) or ( key == "command" ) )
566 continue;
567
568 entry = infoMap [key];
569
570 // item exists in appConfig?
571 if ( appConfig->haveItem (key) )
572 {
573 appConfig->setItemValue ( key, entry );
574 continue;
575 }
576
577 // no, then check fileConfig?
578 if ( fileConfig->haveItem (key) )
579 fileConfig->setItemValue ( key, entry );
580 }
581
582 toGui ();
583 }
584
585}

760 *
761 * This function is called when a timer event arrived. Here timer 0 is
762 * used to load an image after the gui is established.
763 *
764 * \param tEvent
765 *
766 */
767
768void MainWnd::timerEvent ( QTimerEvent * tEvent )
769{
770 int id;
771
772 id = tEvent->timerId ();
773
774 killTimer (id);
775
776 if ( timerId == 0 ) // called from init
777 {
778 if ( QFile::exists (imgName) )
779 {
780 imageWidget->loadImage ();
781 }
782
783 if ( jobMan->getJobCount () > 0 )
784 {
785 timerId = 1;
786 startTimer (200); // activates job manager
787 }
788 }
789
790 if ( timerId == 1 )
791 {
792 jobMan->execJobs (); // job queue is already parsed
793 }
794
795 if ( timerId == 2 )
796 {
797 statusMsg->setText ( "" ); // delete status message
798 }
799
800 if ( timerId == 3 )
801 {
802 msgBox->close ();
803 }
804
805 timerId = -1; // nothing selected
806
807}
