49/*!
50 *
51 * This function calculates a lookup table for a linear curve type for
52 * every color with a a low and high input. Then it merges the color
53 * channels, calculates the gray channel from them and returns them as
54 * cv::Mat.
55 *
56 * \param matIn
57 *
58 * \return cv::Mat
59 *
60 */
61
62cv::Mat ColorCurves::applyCurves ( cv::Mat & matIn )
63{
64 ColorChannel * colChannel;
65 V_CvMat channels;
66 cv::Mat gray;
67 cv::Mat color;
68 cv::Mat lut;
69
70 cv::split ( matIn, channels );
71
72 for ( int i = 0; i < 3; i++ )
73 {
74 colChannel = colorChannels [i];
75
76 colChannel->setChannelMat ( channels [i] );
77
78 colChannel->createLinear ();
79 lut = colChannel->getLookup ();
80
81 cv::LUT ( channels [i], lut, channels [i] );
82 }
83
84 cv::merge ( channels, color );
85
86 cvtColor ( color, gray, COLOR_RGB2GRAY );
87
88 chanGray.setChannelMat (gray);
89
90 return color;
91
92}

95/*!
96 *
97 * This function is called from the tools widget when some curves or
98 * black or white points has changed in the curves dialog. Then a lookup
99 * table is calulated from the curves and applied to the given image mat
100 * and stored in the output image mat.
101 *
102 * When a applyColors is set then the colors channels are individually
103 * calculated and merged, this is only the case when 'Auto level' is
104 * executed.
105 *
106 *
107 * \param matIn Reference to matrix of OpenCV
108 * \param matOut Reference to output matrix after applying LUT
109 *
110 */
111
112void ColorCurves::applyCurve ( cv::Mat & matIn, cv::Mat & matOut )
113{
114 V_CvMat channels;
115 cv::Mat lut;
116 cv::Mat ylum; // luminance of ycrcb
117 cv::Mat ycrcb;
118
119 setImage (matIn);
120
121 if (applyColors)
122 {
123 matOut = applyCurves (matIn);
124 return;
125 }
126
127 chanGray.createSamples ();
128
129 lut = chanGray.getLookup ();
130
131 if (isGray) // only gray scale
132 {
133 cv::LUT ( matIn, lut, matOut );
134 return;
135 }
136
137 // convert RGB to luminance space YCrCb
138 cv::cvtColor ( matIn, ycrcb, cv::COLOR_RGB2YCrCb );
139
140 cv::split ( ycrcb, channels );
141
142 cv::LUT ( channels [0], lut, ylum ); // apply your 1D LUT to luminance only
143 channels [0] = ylum;
144
145 cv::merge ( channels, ycrcb );
146
147 // convert luminance space YCrCb to RGB
148 cv::cvtColor ( ycrcb, matOut, cv::COLOR_YCrCb2RGB );
149
150}

153/*!
154 *
155 * This function calculates the luminosity value from the given 3 channel
156 * color and returns it. The color is in uchar format, therefore it its
157 * converted to floating point value in the range between 0 and 1. For
158 * the calculation of the luminosity the OpenCV formula is used. The
159 * given color is in RGB format.
160 *
161 * There are 2 different formula for the conversion from color to gray:
162 *
163 * OpenCV luminosity formula:
164 * Y = 0.299 R + 0.587 G + 0.114 B
165 *
166 * Gimp luminosity formula:
167 * Y = 0.222 R + 0.717 G + 0.061 B
168 *
169 * \param color Vector for 3 colors in uchar format
170 *
171 * \return float Gray value as floating point
172 *
173 */
174
175float ColorCurves::getLuminosity ( Vec3b color )
176{
177 float gray;
178 bool useGimp;
179
180 useGimp = true;
181
182 if (useGimp)
183 {
184 gray = ( 0.222 * color [0] ) + // red
185 ( 0.717 * color [1] ) + // green
186 ( 0.061 * color [2] ); // blue
187 }
188 else
189 {
190 gray = ( 0.299 * color [0] ) + // red
191 ( 0.587 * color [1] ) + // green
192 ( 0.114 * color [2] ); // blue
193 }
194
195 gray = qMin ( 1.0, gray / 255.0 );
196
197 return gray;
198
199}

241/*!
242 *
243 * This function takes the given image and makes a copy of it. If it is a
244 * color image then the color channels are split and stored in the red,
245 * green and blue channels. The color order is RGB.
246 *
247 * \param image
248 *
249 */
250
251void ColorCurves::setImage ( cv::Mat & image )
252{
253 V_CvMat channels;
254 cv::Mat gray;
255
256 matIn = image.clone (); // make a copy
257
258 imageMat.evalMatInfo (image);
259
260 if ( imageMat.getChannels () == 1 )
261 {
262 isGray = true;
263
264 chanGray.setChannelMat (image);
265 }
266 else
267 {
268 isGray = false;
269
270 cv::split ( image, channels );
271
272 chanRed .setChannelMat ( channels [0] );
273 chanGreen.setChannelMat ( channels [1] );
274 chanBlue .setChannelMat ( channels [2] );
275
276 // convert to gray using the luminosity formula of open cv
277 cvtColor ( image, gray, COLOR_RGB2GRAY );
278
279 chanGray.setChannelMat (gray);
280 }
281
282}
