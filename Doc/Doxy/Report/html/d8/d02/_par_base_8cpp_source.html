252/*!
253 *
254 * This function sets the default label for all items and limits the text
255 * with to 'maxTextWidth'.
256 *
257 */
258
259void ParBase::updateList ()
260{
261 ParItem parItem;
262
263 // set default label
264 for ( int i = 0; i < parList.count (); i++ )
265 {
266 parList.getParItem ( i, parItem );
267 parItem.infos ["label"] = parItem.infos ["info"];
268
269 if ( parItem.varType == "string" )
270 {
271 if ( parItem.infos.contains ( "options" ) )
272 {
273 parItem.toString ();
274 }
275 }
276 else
277 {
278 parItem.infos ["maxWidth"] = QString::number (maxTextWidth);
279 }
280
281 parList.setParItem ( i, parItem );
282 }
