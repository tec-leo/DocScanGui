261/*!
262 *
263 * This function shows the 'content' of the user manual. Therefore it
264 * locates the pdf file and opens it or extracts is from the resources it
265 * it does not exist.
266 *
267 */
268
269void MainBar::helpContents ()
270{
271 QString cwd;
272 QString qrcName;
273 QString name;
274 QFile file;
275
276 cwd = QDir::currentPath ();
277 name = cwd + "/" + cProgramName + ".pdf";
278
279 if ( not QFile::exists (name) )
280 {
281 qrcName = ":/Doc/" + cProgramName + ".pdf";
282 file.setFileName (qrcName);
283
284 if ( not file.open ( QIODevice::ReadOnly ) )
285 {
286 QMessageBox::warning ( this, "Cannot read manual file", name );
287 return;
288 }
289
290 file.copy (name);
291
292 file.close ();
293 }
294
295 QDesktopServices::openUrl ( QUrl ( "file:// /" + name ) );
296
297}
