70/*!
71 *
72 * This function converts the image first to gray scale and then applies
73 * 'threshold' to the simple threshold function of OpenCV. If the given
74 * threshold is below 0 the it uses the 'otsu' algorithm (automatic
75 * threshold finding) to make it binary.
76 *
77 * \param input Matrix to binarize
78 * \param threshold
79 *
80 * \return Mat Resulting matrix, binarized and white on black
81 *
82 */
83
84Mat ImageMat::binarize ( Mat input, int threshold )
85{
86 Mat binImg;
87
88 input = convToGray (input);
89
90 if ( threshold < 0 )
91 cv::threshold ( input, binImg, 0, 255, THRESH_OTSU );
92 else
93 cv::threshold ( input, binImg, threshold, 255, THRESH_BINARY );
94
95 return binImg;
96
97}

100/*!
101 *
102 * This function copies from the source matrix to the destination matrix
103 * at the position of rect in the destination. It uses a roi matrix which
104 * shares the data with the source matrix.
105 *
106 * \param src Source matrix
107 * \param dst Destination matrix
108 * \param rect Position and size in the Destination matrix
109 *
110 */
111
112void ImageMat::copyTo ( Mat src, Mat dst, cv::Rect rect )
113{
114 Mat roiMat;
115 Mat mask;
116 int hDif;
117 int wDif;
118
119 rect.width = src.cols;
120 rect.height = src.rows;
121
122 // check if copy is possible, change rect position if necessary
123 wDif = dst.cols - ( rect.x + rect.width );
124
125 if ( wDif < 0 )
126 rect.x = rect.x + wDif;
127
128 hDif = dst.rows - ( rect.y + rect.height );
129
130 if ( hDif < 0 )
131 rect.y = rect.y + hDif;
132
133 if ( ( rect.x < 0 ) or ( rect.y < 0 ) )
134 {
135 // change size of src and rect?
136 return;
137 }
138
139 // address region of interest in destination matrix
140 roiMat = cv::Mat ( dst, rect );
141
142 mask = src.clone (); // make a copy
143 mask = 1; // and set mask to 1 - copy every pixel
144
145 cv::copyTo ( src, roiMat, mask );
146
147}

150/*!
151 *
152 * This function takes a binary image and counts the black and white
153 * points. When it contains more white pixels than black ones then the
154 * image is inverted.
155 *
156 * \param input Input matrix
157 *
158 * \return Mat Resulting matrix black on white
159 *
160 */
161
162Mat ImageMat::blackOnWhite ( Mat input )
163{
164 int white;
165 int black;
166
167 // count the number of black and white pixels
168 white = countNonZero (input);
169 black = input.size ().area () - white;
170
171 // if the image is mostly white (white background), invert it
172 if ( white > black )
173 input = ~input;
174
175 return input;
176
177}

302void ImageMat::evalMatInfo ( Mat & mat )
303{
304 QString typeStr;
305
306 elemSize = mat.elemSize (); // matrix element size in bytes
307 channels = mat.channels (); // number of channels, typically 3 for a colour image
308 depth = mat.depth (); // pixel type, eg. CV_8U for 8bit unsigned

310

321 steps = mat.step1 ();
322
323 imgSize = cols * rows * channels * pixLen; // size in bytes

328/*!
329 *
330 * This function counts the pixel which are closer to black by a defined
331 * threshold value. It assumes that the given matrix is a gray image (one
332 * channel) with unsigned 8 bit values and white on black text.
333 *
334 * It calculates the ratio of black text to all pixel on a
335 * document. Black text means a value of 0 for the black pixels.
336 *
337 * \param mat Input matrix
338 *
339 * \return double ratio of black text to all pixels in a text doc
340 *
341 */
342
343double ImageMat::getBlackRatio ( Mat & mat )
344{
345 double ratio;
346 int size;
347 int black;
348 int white;
349 Mat srcMat;
350
351 ratio = 0;
352 black = 0;
353 size = mat.rows * mat.cols;
354
355 if ( mat.type () != CV_8U )
356 srcMat = convToGray (mat); // convert to grayscale
357 else
358 srcMat = mat.clone (); // do not change incoming matrix
359
360 threshold ( srcMat, srcMat, 150, 255, 0 );
361
362 white = countNonZero (srcMat);
363 black = size - white;
364
365 ratio = 1.0 * black / size; // for white text on black background
366
367 return ratio;
368
369}

414 *
415 * This function puts every point with a value greater than 0 to a vector
416 * of points. The image must already be binarized and 8 bit unsigned
417 * grayscaled. 'Binarized' means white text (content) on black so that

419 *
420 * \param mat Input matrix

422 * \return V_CvPoint Vector of points
423 *

426V_CvPoint ImageMat::getWhitePoints ( Mat & mat )
427{
428 V_CvPoint points;
429 cv::Point point;
430 uchar val;
431
432 for ( int row = 0; row < mat.rows; row++ )
433 {
434 for ( int col = 0; col < mat.cols; col++ )
435 {
436 val = mat.at < uchar > ( row, col );
437
438 if ( val > 0 )
439 {
440 point.x = col;
441 point.y = row;
442
443 points.push_back (point);
444 }
445 }
446 }
447
448 return points;

590

592
593 depth = type & CV_MAT_DEPTH_MASK;
594 channels = 1 + ( type >> CV_CN_SHIFT );
595
596 if ( depth < matTypeStr.count () )
597 typeStr = matTypeStr [depth];
598 else
599 typeStr = "User";
600
601 text.setNum (channels);
