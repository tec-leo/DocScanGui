618/*!
619 *
620 * This function evaluates the optimal values for the lowest and highest
621 * input value for the given channel. It uses a percentage of a
622 * hysteresis when history values have a 'peak'. This method is taken
623 * from Gimp.
624 *
625 * \param level
626 *
627 */
628
629void CurvesDialog::evalAutoLevel ( ColorChannel * level )
630{
631 cv::Mat * channel;
632 Histogram hist;
633 L_Float histValues;
634 float newCount;
635 float percent;
636 float nextPerc;
637 float bias;
638 float lowInput;
639 float highInput;
640 float histVal;
641 float histNext;
642 float histPrev;
643 int histBins;
644
645 channel = level->getChannelMat ();
646
647 hist.create ( * channel );
648
649 histValues = hist.getNormalized ( "Sum" );
650 histBins = histValues.count () - 1; //- 1 because of for loop
651
652 bias = 0.006f;
653 newCount = 0.0;
654 lowInput = 0.0;
655 highInput = 1.0;
656
657 // set the low input, -1 is already in histBins
658 for ( int i = 0; i < histBins; i++ )
659 {
660 histVal = histValues [i];
661 histNext = histValues [ i + 1 ];
662
663 newCount = newCount + histVal;
664 percent = newCount;
665
666 nextPerc = ( newCount + histNext );
667
668 if ( fabs ( percent - bias ) < fabs ( nextPerc - bias ) )
669 {
670 lowInput = (float) ( i + 1 ) / histBins;
671 break;
672 }
673 }
674
675 newCount = 0.0;
676
677 // set the high input
678 for ( int i = histBins; i > 0; i-- )
679 {
680 histVal = 1.0 * histValues [i];
681 histPrev = 1.0 * histValues [ i - 1 ];
682
683 newCount = newCount + histVal;
684 percent = newCount;
685
686 nextPerc = ( newCount + histPrev );
687
688 if ( fabs ( percent - bias ) < fabs ( nextPerc - bias ) )
689 {
690 highInput = (float) ( i - 1 ) / histBins;
691 break;
692 }
693 }
694
695 // sets the low and high input, the same as black and white point
696 level->setLowHigh ( lowInput, highInput );
697
698}

701/*!
702 *
703 * This function is called from the main window after a mouse press event
704 * happened in image scene. The mouse position is used to evaluate the
705 * color value of the image at this position. The color then defined the
706 * current black or white point value.
707 *
708 * \param pos
709 *
710 */
711
712void CurvesDialog::evalMousePos ( QPointF pos )
713{
714 cv::Vec3b color;
715 cv::Point point;
716 int intensitiy;
717 float value;
718
719 if ( colPicker <= cNoColorPick ) // no color picker active
720 return;
721
722 point.x = pos.x ();
723 point.y = pos.y ();
724
725 if ( ( point.x >= image.cols ) or ( point.y >= image.rows ) )
726 return;
727
728 intensitiy = 0;
729
730 color = getColorAtPoint (point);
731
732 if (isGray)
733 {
734 intensitiy = image.at < uchar > (point);
735
736 value = 1.0 * intensitiy / 255;
737 intensitiy = value * 100.0;
738
739 if ( colPicker == cPickerBlack ) // black point
740 channel->setBlackPnt (value);
741 else
742 channel->setWhitePnt (value);
743 }
744 else
745 {
746 if ( colPicker == cPickerBlack ) // black point
747 curves->setBlackPnt (color);
748
749 if ( colPicker == cPickerWhite ) // white point
750 curves->setWhitePnt (color);
751
752 intensitiy = 100.0 * curves->getLuminosity (color);
753 }
754
755 blockControls (true);
756
757 if ( colPicker == cPickerBlack ) // black point
758 txtBlackPnt->setValue (intensitiy);
759 else
760 txtWhitePnt->setValue (intensitiy);
761
762 blockControls (false);
763
764 // emit applyCurves (); // inform main window to update the image view
765
766 curveWidget->updateChart ();
767
768}

771/*!
772 *
773 * This function evaluates the color of the image at the given mouse
774 * position. It allows to use a rectangle around the point to apply a
775 * simple blur filter before returning the color.
776 *
777 * \param point
778 *
779 * \return cv::Vec3b
780 *
781 */
782
783cv::Vec3b CurvesDialog::getColorAtPoint ( cv::Point point )
784{
785 cv::Mat dstMat;
786 cv::Vec3b color;
787 cv::Rect rect;
788 cv::Size size;
789 Point2f center;
790 int x, y;
791 int dif;
792 int radius;
793
794 if (isGray)
795 color [0] = image.at < uchar > (point);
796 else
797 color = image.at < cv::Vec3b > (point);
798
799 radius = txtRadius->text ().toInt ();
800
801 if ( radius <= 1 )
802 return color;
803
804 dif = radius - 1;
805
806 x = point.x - dif;
807 y = point.y - dif;
808
809 if ( ( ( x - dif ) < 0 ) or ( ( x + dif ) >= image.cols ) or
810 ( ( y - dif ) < 0 ) or ( ( y + dif ) >= image.rows ) )
811 {
812 return color;
813 }
814
815 rect.x = x - dif;
816 rect.y = y - dif;
817
818 rect.width = 2 * dif + 1;
819 rect.height = 2 * dif + 1;
820
821 center = point;
822
823 cv::getRectSubPix ( image, rect.size (), center, dstMat );
824
825 size.height = rect.height;
826 size.width = rect.width;
827
828 cv::blur ( dstMat, dstMat, size );
829
830 point.x = dif;
831 point.y = dif;
832
833 if (isGray)
834 color [0] = dstMat.at < uchar > (point);
835 else
836 color = dstMat.at < cv::Vec3b > (point);
837
838 return color;
839
840}

883/*!
884 *
885 * This function is called when the 'Auto Level' button is pressed. It
886 * uses 'evalAutoLevel' to evaluate the optimal values for the black and
887 * white point for every channel.
888 *
889 */
890
891void CurvesDialog::onAutoLevel ()
892{
893 ChannelInfo info;
894
895 fromGui ();
896
897 curves->setImage (image);
898
899 channel = curves->getChannelGray (); // default channel for dialog
900
901 channel->resetPoints (); // sets to linear default
902
903 if (isGray)
904 {
905 evalAutoLevel (channel);
906
907 channel->getChannelInfo (info);
908
909 txtBlackPnt->setValue ( 100 * info.lowInput );
910 txtWhitePnt->setValue ( 100 * info.highInput );
911
912 curves->setApplyColors (false);
913 }
914 else
915 {
916 channel = curves->getChannelRed ();
917 evalAutoLevel (channel);
