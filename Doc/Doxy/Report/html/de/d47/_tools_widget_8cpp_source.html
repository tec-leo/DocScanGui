419/*!
420 *
421 * This function is called when a job was initiated from the command line
422 * interface or a job file. The tools widget executes the jobs like
423 * running a tools chain or individual tools. It also handles the
424 * excecution more several files or all image files in a directory.
425 *
426 * \param curJob
427 *
428 */
429
430void ToolsWidget::executeJob ( JobInfo curJob )
431{
432 QDir dir;
433 QString name;
434 QString exportDir;
435 L_String dirFiles;
436 L_String files;
437 L_Int toolList;
438 bool force;
439
440 jobInfo = curJob;
441 infoMap = jobInfo.infoMap;
442
443 if ( jobInfo.topic != "tools" )
444 {
445 text = "Topic " + jobInfo.topic + " not known !!";
446 msgLog->showError ( "ToolsWidget", "executeJob", text );
447
448 return;
449 }
450
451 if ( not ( ( jobInfo.command == "run-chain" ) or
452 ( jobInfo.command == "run-tools" ) ) )
453 {
454 text = "Command " + jobInfo.command + " not known !!";
455 msgLog->showError ( "ToolsWidget", "executeJob", text );
456
457 return;
458 }
459
460 cmdMode = true; // used when loading images from a list or directory
461 force = true; // force tools even when not selected
462
463 // check export dir
464 if ( infoMap.contains ( "export-dir" ) )
465 exportDir = infoMap [ "export-dir" ];
466 else
467 exportDir = imageWidget->getExportDir ();
468
469 if ( not fileUtils.makePath (exportDir) )
470 {
471 text = "Cannot create export directory : " + exportDir +
472 ", using current export directory !!";
473
474 msgLog->showError ( "ToolsWidget", "executeJob", text );
475
476 exportDir = imageWidget->getExportDir ();
477 }
478
479 // check image files
480 if ( infoMap.contains ( "image-files" ) )
481 {
482 name = infoMap [ "image-files" ];
483
484 files = name.split ( "," );
485 }
486
487 // check image directory after files are evaluated
488 if ( infoMap.contains ( "image-dir" ) )
489 {
490 name = infoMap [ "image-dir" ];
491 dir.setPath (name);
492
493 dirFiles = dir.entryList ( QDir::Files | QDir::NoDotAndDotDot );
494
495 for ( int i = 0; i < dirFiles.count (); i++ )
496 {
497 text = name + "/" + dirFiles [i];
498 files.append (text);
499 }
500 }
501
502 if ( jobInfo.command == "run-tools" )
503 {
504 text = infoMap [ "tools" ];
505 toolList = getToolsList (text);
506 force = true; // force tools even when not selected
507 }
508
509 if ( jobInfo.command == "run-chain" )
510 {
511 text = infoMap [ "tools" ];
512 toolList = runList;
513 force = false; // only selected tools are executed
514 }
515
516 if ( files.count () > 0 )
517 {
518 for ( int i = 0; i < files.count (); i++ )
519 {
520 imageName = files [i];
521 imageWidget->setImageName (imageName);
522
523 if ( not QFile::exists (imageName) )
524 continue;
525
526 loadImage (); // here it is checked if the image was loaded
527
528 if ( not imgLoaded )
529 continue;
530
531 runChain ( toolList, force );
532
533 name = exportDir + "/" + fileUtils.getBaseName ( imageName, true );
534 saveImage (name);
535 }
536 }
537 else
538 runChain ( toolList, force );
539
540}

543/*!
544 *
545 * This function takes a comma separated string of short tool names and
546 * fills a integer list with the indexes of the tools.
547 *
548 * \param tools
549 *
550 * \return L_Int
551 *
552 */
553
554L_Int ToolsWidget::getToolsList ( QString tools )
555{
556 L_String jobNames;
557 L_Int jobTools;
558 int jobIdx;
559
560 jobNames = tools.split ( "," );
561
562 for ( int idx = 0; idx < jobNames.count (); idx++ )
563 {
564 text = jobNames [idx];
565
566 jobIdx = jobList.indexOf (text);
567
568 if ( jobIdx >= 0 )
569 jobTools.append (jobIdx);
570 else
571 {
572 text = "Tools " + text + " not found !!";
573 msgLog->showError ( "ToolsWidget", "getToolsList", text );
574 }
575 }
576
577 return jobTools;
578
579}

653/*!
654 *
655 * This function converts the given Mat to the format for loading defined
656 * in fileConfig. Any alpha channel is removed, the color order is
657 * changed from BGR to RGB. Only the 8 bit unsigned image format is
658 * supported. Floating point formats are converted to 8 bit unsigned.
659 *
660 * \param mat Input matrix
661 *
662 */
663
664void ToolsWidget::normInputImg ( Mat & mat )
665{
666
667 imageMat.removeAlpha (mat); // remove alpha
668
669 if ( mat.channels () == 3 ) // change color order
670 cvtColor ( mat, mat, COLOR_BGR2RGB );
671
672 if ( mat.depth () != CV_8U )
673 mat = imageMat.convTo8bit (mat);
674
675 if ( fileConfig->loadGray )
676 mat = imageMat.convToGray (mat);
677
678}

681/*!
682 *
683 * This function converts the given Mat to the format defined for saving
684 * in fileConfig. The color order is changed back from RGB to BGR for
685 * color images. The image is saved as png or jpg with the definded
686 * compression level.
687 *
688 * \param mat Input matrix
689 *
690 * \return Mat
691 *
692 */
693
694Mat ToolsWidget::normOutputImg ( Mat mat )
695{
696 Mat normOut;
697
698 if ( mat.channels () == 3 ) // change color order
699 {
700 cvtColor ( mat, normOut, COLOR_RGB2BGR );
701 }
702 else
703 {
704 normOut = mat.clone ();
705 }
706
707 return normOut;
708
709}

712/*!
713 *
714 * This function loads the image from file if it exists and shows the
715 * image info in the status bar if not called from the jobmanager.
716 *
717 * \return bool
718 *
719 */
720
721bool ToolsWidget::loadImage ()
722{
723
724 clearHistory ();
725
726 imageName = imageWidget->getImageName ();
727
728 matIn = cv::imread ( ToString (imageName), IMREAD_UNCHANGED );
729
730 imgLoaded = not matIn.empty ();
731
732 if ( matIn.empty () ) // cannot be loaded
733 return false;
734
735 normInputImg (matIn);
736
737 matOrg = matIn.clone ();
738
739 // store curves input before running chain, will be updated in chain
740 matMid = matOrg.clone ();
741 matIn = matOrg.clone ();
742 matOut = matOrg.clone (); // must always be set if the chain is not run
743
744 if ( not cmdMode )
745 showImageInfo (matOrg);
746
747 return true;
748
749}

752/*!
753 *
754 * This function saves the image with the given file name (with complete
755 * path). The image is normalized before this means the color order is
756 * changed back from RGB to BGR for color images. Also different output
757 * formats (jpg, png) are checked before saving.
758 *
759 * \param newName
760 *
761 */
762
763void ToolsWidget::saveImage ( QString newName )
764{
765 L_String parts;
766 QString suffix;
767 QString saveName;
768 Mat saveMat;
769 int jpgComp;
770
771 imageName = newName;
772
773 saveMat = normOutputImg (matOut);
774
775 if ( fileConfig->saveAsIs )
776 {
777 cv::imwrite ( ToString (imageName), saveMat );
778 return;
779 }
780
781 parts = fileUtils.splitFileName (imageName);
782 suffix = parts [2];
783
784 if ( fileConfig->saveType == "png" )
785 {
786 saveName = parts [0] + "/" + parts[1] + ".png";
787 cv::imwrite ( ToString (saveName), saveMat );
788 }
789 else
790 {
791 saveName = parts [0] + "/" + parts[1] + ".jpg";
792
793 jpgComp = fileConfig->saveJpgComp;
794 cv::imwrite ( ToString (saveName), saveMat,
795 { cv::ImwriteFlags::IMWRITE_JPEG_QUALITY, jpgComp } );
796 }
797
798}

1100/*!
1101 *
1102 * This function applies a perspective transformation to the input image.
1103 * The function warpPerspective transforms the source image using a
1104 * transformation matrix. This matrix is calculated with
1105 * getPerspectiveTransform which needs the coordinates of the quadrangle
1106 * vertices in the source image and the corresponding quadrangle vertices
1107 * in the destination image.
1108 *
1109 */
1110
1111void ToolsWidget::runDeskew ()
1112{
1113 L_PointsF points;
1114 V_CvPoint2f srcPnts;
1115 V_CvPoint2f dstPnts;
1116 cv::Mat transform;
1117 float widthA;
1118 float widthB;
1119 float heightA;
1120 float heightB;
1121 float maxWidth;
1122 float maxHeight;
1123
1124 points = imageScene->getPath ();
1125
1126 if ( points.count () != 4 )
1127 return;
1128
1129 srcPnts.resize (4);
1130 dstPnts.resize (4);
1131
1132 for ( int i = 0; i < 4; i++ )
1133 {
1134 srcPnts [i].x = points [i].x ();
1135 srcPnts [i].y = points [i].y ();
1136 }
1137
1138 std::swap ( srcPnts [2], srcPnts [3] );
1139
1140 // compute the width of the new image, top and bottom
1141 widthA = graphUtils.getDistance ( points [0], points [1] );
1142 widthB = graphUtils.getDistance ( points [2], points [3] );
1143 maxWidth = qMax ( widthA, widthB );
1144
1145 // compute the height of the new image, left and right
1146 heightA = graphUtils.getDistance ( points [0], points [2] );
1147 heightB = graphUtils.getDistance ( points [1], points [3] );
1148 maxHeight = qMax ( heightA, heightB );
1149
1150 // set the destination points of the image
1151 // order of points in OpenCV functions is
1152 // top-left, top-right, bottom-right, and bottom-left
1153
1154 dstPnts [0].x = 0;
1155 dstPnts [0].y = 0;
1156
1157 dstPnts [1].x = maxWidth - 1;
1158 dstPnts [1].y = 0;
1159
1160 dstPnts [2].x = maxWidth - 1;
1161 dstPnts [2].y = maxHeight - 1;
1162
1163 dstPnts [3].x = 0;
1164 dstPnts [3].y = maxHeight - 1;
1165
1166 // compute the perspective transform matrix and then apply it
1167 transform = getPerspectiveTransform ( srcPnts, dstPnts );
1168
1169 warpPerspective ( matIn, matOut, transform, cv::Size2f ( maxWidth, maxHeight ) );
1170
1171}
