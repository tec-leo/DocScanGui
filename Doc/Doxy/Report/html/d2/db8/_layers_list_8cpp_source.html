267/*!
268 *
269 * This function is called when a button in the widget is pressed. It is
270 * checked if the sender is a QPushButton and if so a signal sent out
271 * which informs about label and pressed state of this button. It is not
272 * necessary to block signals when calling setChecked, because it does
273 * not signal 'clicked', but 'toggled'.
274 *
275 * \param checked Checked state of the selected button
276 *
277 */
278
279void LayersList::onBtnClicked ( bool checked )
280{
281 int newIdx;
282
283 // QObject::sender () returns the object which sent the signal
284 iconBtn = qobject_cast < IconButton * > ( QObject::sender () );
285
286 if (iconBtn)
287 {
288 newIdx = iconBtn->property ( "Index" ).toInt ();
289 checked = iconBtn->isChecked ();
290
291 // always select current button (pressed and on)
292 if ( not checked )
293 {
294 iconBtn->setChecked (true);
295 }
296
297 if ( newIdx != curIndex )
298 {
299 iconBtn = buttons [curIndex];
300
301 iconBtn->setChecked (false); // deselect previous button
302
303 curIndex = newIdx;
304
305 emit buttonClicked ( curIndex, checked );
306 }
307 }
308
309}

312/*!
313 *
314 * This function is called when a button in the widget is pressed. It is
315 * checked if the sender is a QPushButton. It toggles the 'active'
316 * property of this button and changes the icon depending on the 'active'
317 * state. Index and 'active' state are then sent by the signal
318 * 'btnIconToggled'.
319 *
320 * \param index 'Active' state (property) of the selected button
321 *
322 */
323
324void LayersList::onIconToggled ( int index )
325{
326 bool active;
327
328 iconBtn = buttons [index];
329
330 if (iconBtn)
331 {
332 active = not iconBtn->property ( "Active" ).toBool ();
333 iconBtn->setProperty ( "Active", QVariant (active) );
334
335 if (active)
336 iconBtn->setIcon (btnEnabled);
337 else
338 iconBtn->setIcon (btnDisabled);
339 }
340
341 emit btnIconToggled ( index, active );
342
343}
