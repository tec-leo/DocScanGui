80/*!
81 *
82 * This function calculates a histogram of a the given channel with
83 * 'calcHist' from OpenCV. The histogram is in the range from 0 to 256
84 * and uniform .
85 *
86 */
87
88void Histogram::calcHistogram ()
89{
90 bool accumulate; // true - histogram is not cleared in the beginning
91 bool uniform; // histogram is uniform or not
92 float histY;
93
94 uniform = true; // set histogram param
95 accumulate = false; // clear histogram in the beginning
96
97 // set the ranges ( for B, G, R )
98 float range [] = { 0, 256 }; // the upper boundary is exclusive
99 const float * histRange [] = {range};
100 calcHist ( & matImg, 1, 0, Mat (), matHist, 1, & numBins,
101 histRange, uniform, accumulate );
102
103 histValues.clear ();
104
105 maxValue = 0;
106 histSum = 0;
107
108 for ( int i = 0; i < numBins; i++ )
109 {
110 histY = matHist.at < float > (i);
111
112 histValues.append (histY);
113 histSum = histSum + histY;
114
115 maxValue = qMax ( maxValue, histY );
116 }
117
118}

121/*!
122 *
123 * This function returns the normalized histogram values, they can be
124 * normalized to the maximum value or to the sum of all histogram values.
125 *
126 * \param type
127 *
128 * \return L_Float

131
132L_Float Histogram::getNormalized ( QString type )
133{
134 L_Float normValues;
135 float curValue;
136
137 normValues.resize (numBins);
138
139 for ( int i = 0; i < numBins; i++ )
140 {
141 curValue = histValues [i];
142
143 if ( type == "Max" )
144 curValue = curValue / maxValue;
145 else
146 curValue = curValue / histSum;
147
148 normValues [i] = curValue;
149 }
150
151 return normValues;
