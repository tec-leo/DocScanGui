102
103 fileName = name;

107 fileConfig = new FileConfig ();
108
109 parData.append (appConfig);

111

113
114 // update all parameter classes with the new parameter file

116
117 configMRU.load ( cConfigMRU, "MRUFile", 10 );

119 connect ( & configMRU, & HistoryList::itemSelected,
120 this, & ParManager::onLoadMRU );
121
122 if ( QFile::exists (fileName) )
123 return true;
124 else
125 return false;

342/*!
343 *
344 * This function creates a backup parameter file and saves all
345 * parameters. The appCurves and mainWnd are handled separately because
346 * they are no base parameter class.
347 *
348 * \param createBackup
349 *
350 * \return bool
351 *
352 */
353
354bool ParManager::save ( bool createBackup )
355{
356 ParBase * parBase;
357 QString help;
358 QString tmp;
359 QString suffix;
360 QString oldName;
361 QString baseName;
362 int pos;
363
364 if (createBackup)
365 {
366 pos = fileName.lastIndexOf ( '.' );
367 baseName = fileName.left (pos);
368 suffix = fileName.mid (pos);
369
370 // store a backup file
371 for ( int i = 0; i < 10000; i++ )
372 {
373 help = QString ( "(%1)" ).arg ( i + 1, 2, 10, QChar ( '0' ) );
374 tmp = baseName + help + suffix;
375
376 if ( not QFile::exists (tmp) )
377 {
378 QFile::copy ( fileName, tmp );
379 break;
380 }
381 }
382 }
383
384 appCurves->save ();
385
386 configMRU.append (fileName);
387 configMRU.save ();
388
389 mainWnd->saveConfig ();
390
391 parFile->setFileName (fileName);
392
393 parFile->setSection (cParSection);
394
395 for ( int i = 0; i < parData.count (); i++ )
396 {
397 parBase = parData.at (i);
398 parBase->save ();
399 }
400
401 return parFile->save (); // write to file
402
403}
