360/*!
361 *
362 * This function is called from the drop combo box when the selected job
363 * file has changed.
364 *
365 * \param fileName
366 *
367 */
368
369void JobFilesDlg::onJobsChanged ( QString fileName )
370{
371 QString name;
372 QString text;
373
374 name = fileName;
375 name = jobFiles->getFileName ();
376
377 jobView->clear (); // empty text edit window
378
379 if ( not QFile::exists (name) )
380 {
381 fileUtils.createFile (name);
382 return;
383 }
384 else
385 fileUtils.setFileName (name);
386
387 fileUtils.readFile (text);
388
389 jobView->setText (text);
390
391 if ( name != jobFile )
392 {
393 fileWatch->removePath (jobFile);
394
395 jobFile = name; // store name
396
397 fileWatch->addPath (name);
398 }
