232/*!
233 *
234 * This function rounds the internal value with the increment and limits
235 * to min / max limits if activated.
236 *
237 * \param round
238 *
239 */
240
241void ParItem::roundLimit ( bool round )
242{
243 // round it to the increment of the parameter
244 if (round)
245 value = VarUtils::round ( value / incr ) * incr;
246
247 // limit to minimum and maximum values
248 if ( ( value > rangeMax ) and limitMax )
249 value = rangeMax;
250
251 if ( ( value < rangeMin ) and limitMin )
252 value = rangeMin;
253
254}

257/*!
258 *
259 * This function dereferences value pointer 'ptrVar' and converts it to a
260 * double. It does also a rounding of the value to the increments if
261 * round is activated.
262 *
263 * \param round
264 *
265 * \return double Final value, rounded or not
266 *
267 */
268
269double ParItem::get ( bool round )
270{
271 double outVal;
272
273 value = * (double *) ptrVar;
274
275 outVal = value;
276
277 if (round)
278 outVal = VarUtils::round ( outVal / incr ) * incr;
279
280 return outVal;
281
282}
