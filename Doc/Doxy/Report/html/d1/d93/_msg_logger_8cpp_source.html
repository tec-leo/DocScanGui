81/*!
82 *
83 * This function combines the messages of the given message infos to a
84 * string list with a separation of the message type and returns it.
85 *
86 * \return L_String
87 *
88 */

90L_String MsgLogger::getLogList ()
91{
92 L_String msgList;
93 MsgInfo msgInfo;
94 QString msg;
95
96 msgList.clear ();
97
98 if ( errors.count () > 0 )
99 {
100 msg = "\nErrors found :";
101 msgList.append (msg);
102
103 msgList.append ( getLogList (errors) );
104 }
105
106 if ( warnings.count () > 0 )
107 {
108 msg = "\nWarnings found :";
109 msgList.append (msg);
110
111 msgList.append ( getLogList (warnings) );
112 }
113
114 if ( infos.count () > 0 )
115 {
116 msg = "\nInfos found :";
117 msgList.append (msg);
118
119 msgList.append ( getLogList (infos) );
120 }
121
122 return msgList;
123
124}

127/*!
128 *
129 * This function combines the messages of the given message infos to a
130 * string list and returns it.
131 *
132 * \param infos
133 *
134 * \return L_String
135 *
136 */
137
138L_String MsgLogger::getLogList ( L_MsgInfos & infos )
139{
140 L_String msgList;
141 MsgInfo msgInfo;
142 QString msg;
143
144 msgList.clear ();
145
146 for ( int i = 0; i < infos.count (); i++ )
147 {
148 msgInfo = infos [i];
149
150 msg = msgInfo.title + " : " + msgInfo.text;
151
152 if ( not msgInfo.info.isEmpty () )
153 msg = msg + " " + msgInfo.info;
154
155 msgList.append (msg);
156 }
157
158 return msgList;
159
160}
