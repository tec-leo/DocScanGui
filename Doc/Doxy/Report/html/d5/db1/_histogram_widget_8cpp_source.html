79/*!
80 *
81 * This function adds the values of the calculated histogram to the given
82 * BarSet. The values are normalized to the number of pixels in the
83 * image.
84 *
85 * \param barSet
86 *

88
89void HistogramWidget::buildBarSet ( QBarSet * barSet )
90{
91 float histY;
92 L_Float histValues;
93 int pixels;
94
95 histValues = histogram.getValues ();
96 pixels = histogram.getPixels ();
97
98 for ( int i = 0; i < histValues.count (); i++ )
99 {
100 // normalize to 100 percent
101 histY = 100.0 * histValues [i] / pixels;
102
103 barSet->append (histY);
104 }
105
106}

137/*!
138 *
139 * This function creates the bar chart from the calculated cv mat and
140 * adds the values to the series with every value as own category. It
141 * creates a set of bar sets and adds it to the bar series which takes
142 * the ownership of the bar sets.
143 *
144 */
145
146void HistogramWidget::createChart ()
147{
148 QValueAxis * axisX;
149 QValueAxis * axisY;
150 QBarSet * barSet;
151
152 barSeries = new QBarSeries ();
153 barChart->removeAllSeries ();
154
155 barSet = new QBarSet ( "1" ); // first bar set
156 barSet->setPen ( QPen ( Color::blue ) );
157 barSet->setColor ( Color::lightblue );
158
159 buildBarSet (barSet);
160
161 barSeries->append (barSet); // series takes the ownership of the barSet
162 barChart->addSeries (barSeries);
163 barChart->setTitle ( "Histogram" );
164 barChart->setAnimationOptions ( QChart::SeriesAnimations );
165
166 axisX = new QValueAxis ();
167 axisX->setRange ( 1, 100 );
168 axisX->setTickCount (10);
169 axisX->applyNiceNumbers ();
170
171 axisY = new QValueAxis ();
172 axisY->setRange ( 1, 100.0 );
173 axisY->setTickCount (10);
174 axisY->applyNiceNumbers ();
175
176 barSeries->attachAxis (axisX);
177 barSeries->attachAxis (axisY);
178
179 barChart->addAxis ( axisX, Qt::AlignBottom );
180 barChart->addAxis ( axisY, Qt::AlignLeft );
181
182 barChart->legend ()->setVisible (false);
183
184}
